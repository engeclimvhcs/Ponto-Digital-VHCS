const URL_GOOGLE_SHEETS = "SUA_URL_DO_GOOGLE_APPS_SCRIPT_AQUI";

let streamDeVideo = null;

let modoCameraAoVivo = false;


function switchTab(tab) {

document.querySelectorAll('.content').forEach(c => c.classList.remove('active'));

document.querySelectorAll('.tab-btn').forEach(b => b.classList.remove('active'));

// GPS OTIMIZADO: Converte em Endereço Nominal Completo por Extenso

function obterLocalizacao() {

if ("geolocation" in navigator) {

navigator.geolocation.getCurrentPosition(

(position) => {

dadosPonto.lat = position.coords.latitude;

dadosPonto.lng = position.coords.longitude;

fetch('https://nominatim.openstreetmap.org/reverse?format=json&lat=' + dadosPonto.lat + '&lon=' + dadosPonto.lng + '&zoom=18&addressdetails=1', {

headers: { 'Accept-Language': 'pt-BR' }

})

.then(response => response.json())

.then(data => {

dadosPonto.endereco = data.display_name || "Endereço Urbano Localizado";

document.getElementById('badgeGps').className = "badge badge-success";

document.getElementById('badgeGps').innerText = "OK";

document.getElementById('txtGps').innerHTML = "📍 <b>Endereço:</b> " + dadosPonto.endereco;

validarFormulario();

})

.catch(() => {

dadosPonto.endereco = "Localização Georeferenciada (" + dadosPonto.lat.toFixed(5) + ", " + dadosPonto.lng.toFixed(5) + ")";

document.getElementById('badgeGps').className = "badge badge-success";

document.getElementById('badgeGps').innerText = "OK";

document.getElementById('txtGps').innerHTML = "📍 <b>GPS Ativo:</b> Endereço Detectado";

validarFormulario();

});

},

() => {

document.getElementById('txtGps').innerHTML = "<b style='color:var(--danger)'>Erro: Ative o GPS nas configurações.</b>";

},

{ enableHighAccuracy: true, timeout: 20000, maximumAge: 0 }

);

}

}

window.addEventListener('DOMContentLoaded', obterLocalizacao);


setInterval(() => {

const ago = new Date();

const agora = new Date(ago.getTime() - (3 * 60 * 60 * 1000)); // Ajuste fuso

dadosPonto.dataHora = ago.toLocaleString('pt-BR');

const ano = ago.getFullYear();

const mes = String(ago.getMonth() + 1).padStart(2, '0');

const dia = String(ago.getDate()).padStart(2, '0');

dadosPonto.dataSimples = ano + "-" + mes + "-" + dia;

document.getElementById('txtTempo').innerText = dadosPonto.dataHora;

}, 1000);


function aplicarMascaraCPF(input) {

let v = input.value.replace(/\D/g, "");

if (v.length > 9) {

v = v.replace(/(\d{3})(\d{3})(\d{3})(\d{1,2})/, "$1.$2.$3-$4");

} else if (v.length > 6) {

v = v.replace(/(\d{3})(\d{3})(\d{1,3})/, "$1.$2.$3");

} else if (v.length > 3) {

v = v.replace(/(\d{3})(\d{1,3})/, "$1.$2");

}

input.value = v;

validarFormulario();

}


function validarCPF(cpf) {

cpf = cpf.replace(/[^\d]+/g, '');

if (cpf == '' || cpf.length != 11 || /^(\d)#{10}$/.test(cpf)) return false;

let add = 0;

for (let i = 0; i < 9; i++) add += parseInt(cpf.charAt(i)) * (10 - i);

let rev = 11 - (add % 11);

if (rev == 10 || rev == 11) rev = 0;

if (rev != parseInt(cpf.charAt(9))) return false;

add = 0;

for (let i = 0; i < 10; i++) add += parseInt(cpf.charAt(i)) * (11 - i);

rev = 11 - (add % 11);

if (rev == 10 || rev == 11) rev = 0;

if (rev != parseInt(cpf.charAt(10))) return false;

return true;

}


function checarDuplicidadeHoje(cpf) {

let limpoCpf = cpf.replace(/[^\d]+/g, '');

let historico = JSON.parse(localStorage.getItem('historico_pontos')) || [];

return historico.some(p => p.cpf.replace(/[^\d]+/g, '') === limpoCpf && p.dataSimples === dadosPonto.dataSimples);

}


async function gerenciarCapturaFoto() {

const video = document.getElementById('video');

const btn = document.getElementById('btnFotoUnico');


if (!modoCameraAoVivo && !dadosPonto.foto) {

try {

streamDeVideo = await navigator.mediaDevices.getUserMedia({ video: { facingMode: "user" }, audio: false });

video.srcObject = streamDeVideo;

video.style.display = "block";

document.getElementById('fotoPreview').style.display = "none";

btn.innerText = "Tirar Foto 📸";

btn.className = "btn btn-primary";

modoCameraAoVivo = true;

} catch {

document.getElementById('cameraNativa').click();

}

} else if (modoCameraAoVivo) {

const canvas = document.createElement('canvas');

canvas.width = video.videoWidth || 640;

canvas.height = video.videoHeight || 480;

canvas.getContext('2d').drawImage(video, 0, 0, canvas.width, canvas.height);

salvarFoto(canvas.toDataURL('image/jpeg', 0.5));

if (streamDeVideo) streamDeVideo.getTracks().forEach(t => t.stop());

video.style.display = "none";

btn.innerText = "Refazer Foto 🔄";

btn.className = "btn btn-secondary";

modoCameraAoVivo = false;

} else {

dadosPonto.foto = "";

document.getElementById('badgeCam').className = "badge badge-danger";

document.getElementById('badgeCam').innerText = "OFF";

document.getElementById('txtCam').innerText = "Selfie pendente";

gerenciarCapturaFoto();

}

}


function carregarFotoNativa(input) {

if (input.files && input.files) {

const reader = new FileReader();

reader.onload = (e) => {

salvarFoto(e.target.result);

document.getElementById('btnFotoUnico').innerText = "Refazer Foto 🔄";

};

reader.readAsDataURL(input.files);

}

}


function salvarFoto(base64) {

dadosPonto.foto = base64;

const preview = document.getElementById('fotoPreview');

preview.src = base64;

preview.style.display = "block";

document.getElementById('badgeCam').className = "badge badge-success";

document.getElementById('badgeCam').innerText = "OK";

document.getElementById('txtCam').innerText = "Foto anexada.";

validarFormulario();

}


function validarFormulario() {

const nome = document.getElementById('nome').value.trim();

const cpf = document.getElementById('cpf').value.trim();

const btn = document.getElementById('btnEnviar');

const divErroCpf = document.getElementById('erroCpf');

const divErroDup = document.getElementById('erroDuplicado');

divErroCpf.style.display = "none";

divErroDup.style.display = "none";


let cpfValido = false;

if(cpf.length === 14) {

if(validarCPF(cpf)) {

cpfValido = true;

} else {

divErroCpf.style.display = "block";

}

}


let duplicado = false;

if(cpfValido) {

if(checarDuplicidadeHoje(cpf)) {

duplicado = true;

divErroDup.style.display = "block";

}

}


if (nome && cpfValido && !duplicado && dadosPonto.foto && dadosPonto.lat) {

btn.disabled = false;

} else {

btn.disabled = true;

}

}

if(document.getElementById('senhaAdmin').value === "051116mm") {

document.getElementById('adminAuth').style.display = "none";

document.getElementById('adminData').style.display = "block";

renderInitializeDate();

renderizarPontosAdmin();

} else {

alert("Senha incorreta!");

}

}


function renderInitializeDate() {

const hoje = new Date();

const ano = hoje.getFullYear();

const mes = String(hoje.getMonth() + 1).padStart(2, '0');

const dia = String(hoje.getDate()).padStart(2, '0');

document.getElementById('filtroData').value = ano + "-" + mes + "-" + dia;

}


function enviarPonto(e) {

e.preventDefault();

dadosPonto.nome = document.getElementById('nome').value.trim();

dadosPonto.cpf = document.getElementById('cpf').value.trim();


const btn = document.getElementById('btnEnviar');

btn.disabled = true;

btn.innerText = "Sincronizando Ponto...";


let historico = JSON.parse(localStorage.getItem('historico_pontos')) || [];

historico.push(Object.assign({}, dadosPonto));

localStorage.setItem('historico_pontos', JSON.stringify(historico));


if(URL_GOOGLE_SHEETS.includes("SUA_URL_DO_GOOGLE")) {

alert("Ponto gravado localmente no dispositivo!\n\nAgora o painel admin está corrigido para visualização.");

location.reload();

return;

}


fetch(URL_GOOGLE_SHEETS, {

method: "POST",

mode: "no-cors",

headers: { "Content-Type": "application/json" },

body: JSON.stringify(dadosPonto)

}).then(() => {

alert("Ponto eletrônico transmitido com sucesso!");

location.reload();

}).catch(() => {

alert("Erro na transmissão. Salvo localmente.");

location.reload();

});

}


function renderizarPontosAdmin() {

const filtroData = document.getElementById('filtroData').value;

const filtroNome = document.getElementById('filtroNome').value.trim().toLowerCase();

const lista = document.getElementById('listaPontos');

let historico = JSON.parse(localStorage.getItem('historico_pontos')) || [];

let filtrados = historico.filter(p => {

const matchesData = filtroData ? p.dataSimples === filtroData : true;

const matchesNome = filtroNome ? p.nome.toLowerCase().includes(filtroNome) : true;

return matchesData && matchesNome;

});


if(filtrados.length === 0) {

lista.innerHTML = "<p style='text-align:center; color:var(--text-muted); margin-top:20px;'>Nenhum registro encontrado para este filtro.</p>";

return;

}


lista.innerHTML = "";

for (let i = 0; i < filtrados.length; i++) {

let p = filtrados[i];

let cardHtml = '<div class="card-ponto">' +

'<div class="dados-prestador">' +

'<b>👤 Prestador:</b> ' + p.nome + '<br>' +

'<b>🪪 CPF:</b> ' + p.cpf + '<br>' +

'<b>📅 Horário do Registro:</b> ' + p.dataHora + '<br>' +

'<b>📍 Endereço por Extenso:</b><br>' +

'<span style="font-size:12px; color:var(--text-muted); display:block; margin-bottom: 4px;">' + p.endereco + '</span>' +

'<b>🗺️ Rastreamento:</b> <a class="link-mapa" href="https://www.google.com/maps/search/?api=1&query=' + p.lat + ',' + p.lng + '" target="_blank">Abrir no Google Maps</a>' +

'</div>' +

'<img src="' + p.foto + '" class="foto-large-admin" alt="Foto do Prestador">' +

'</div>';

lista.innerHTML += cardHtml;

}

}
